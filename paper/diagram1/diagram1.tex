\begin{tikzpicture}
% rows: titles at 0, cards at -7mm, -16mm, -25mm
\node[inputc] (source) at (0,-12mm) {Pyomo model\\(source code)};
\node[inputc] (data)   at (0,-25mm) {User-provided\\data};
\node[hot]  (build)     at (35mm,-9.5mm)  {Build Pyomo model};
\node[card] (translate) at (35mm,-18.5mm) {Translate to Gurobi};
\node[card] (solve)     at (35mm,-27.5mm) {Solve with Gurobi};
\node[decide] (dec) at (62mm,-27.5mm) {Feasible?};
\node[ok]  (sol)   at (86mm,-9.5mm)  {Receive solution};
\node[err] (nosol) at (86mm,-27.5mm) {Infeasible or\\unbounded};
\coordinate (top) at (0,-1.5mm); \coordinate (bot) at (0,-32mm);
\begin{scope}[on background layer]
\node[panel, fit=(source)(data)(source|-top)(data|-bot), inner ysep=0pt] (P1) {};
\node[panel, fit=(build)(solve)(build|-top)(solve|-bot), inner ysep=0pt] (P2) {};
\node[panel, fit=(dec)(sol)(nosol)(sol|-top)(nosol|-bot), inner ysep=0pt] (P3) {};
\end{scope}
\node[ptitle] at ([xshift=3pt,yshift=-2.5pt]P1.north west) {Writing code};
\node[ptitle] at ([xshift=3pt,yshift=-2.5pt]P2.north west) {Running code};
\node[ptitle] at ([xshift=3pt,yshift=-2.5pt]P3.north west) {Output \& feedback};
% forward flow
\coordinate (j) at ($(P1.east)!0.5!(P2.west)$);
\draw[flow] (source.east) -| (j |- build) -- (build.west);
\draw[flow, -] (data.east) -| (j |- build);
\draw[flow] (build) -- (translate);
\draw[flow] (translate) -- (solve);
\draw[flow] (solve) -- (dec);
\draw[flow] (dec.north) |- node[blab, pos=0.25, left] {yes} (sol.west);
\draw[flow] (dec) -- node[blab, above] {no} (nosol);
% feedback loops
\draw[loop] (nosol.east) -| ([xshift=4mm]P3.east |- top) |- ([yshift=4mm]P2.north) -| node[lab, pos=0.1] {iterate model} ([xshift=-4mm]P1.west |- source) -- (source.west);
\draw[loop] (nosol.south) |- node[lab, pos=0.75] {iterate data} ([yshift=-4mm]P2.south) -| (data.south);
\end{tikzpicture}
